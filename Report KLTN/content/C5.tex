\chapter{KẾT LUẬN VÀ HƯỚNG PHÁT TRIỂN}
\label{chap:chapter5}

\section{Tổng kết các kết quả đạt được}
\label{sec:tongKet}

Khóa luận đã hoàn thành bốn mục tiêu nghiên cứu đề ra:

\begin{enumerate}
    \item Xây dựng môi trường mô phỏng: phát triển thành công một môi
    trường mô phỏng theo chuẩn Gymnasium \parencite{towers2024gymnasium}, mô
    hình hóa đầy đủ hệ thống $10$ kho, $30$ mặt hàng đại diện ($300$ cặp tác
    tử), với nhu cầu ngẫu nhiên, thời gian giao hàng biến động $[1, 3]$ ngày,
    dung lượng kho hiệu chỉnh riêng theo từng kho và năm nhóm chi phí vận
    hành. Tham số nhu cầu và giá bán được hiệu chỉnh từ bộ dữ liệu M5 Walmart
    \parencite{makridakis2022background}.

    \item Thiết kế hàm phần thưởng: đề xuất hàm phần thưởng phân rã
    cục bộ (dựa trên nguyên lý phần thưởng chênh lệch \parencite{wolpert2001})
    kèm thành phần phạt fill rate ràng buộc theo tinh thần Lagrangian trên cửa
    sổ trượt $30$ ngày, cùng cơ chế chuẩn hóa phần thưởng theo từng cặp giúp
    một mạng nơ-ron duy nhất học đồng thời trên các cặp có quy mô chi phí
    chênh lệch nhiều bậc độ lớn.

    \item Cài đặt và huấn luyện IPPO: triển khai thuật toán Independent
    Multi-Agent PPO \parencite{dewitt2020ippo} với hai mạng Actor và Critic
    độc lập chia sẻ tham số giữa các tác tử \parencite{gupta2017}, sử dụng
    GAE \parencite{schulman2016gae}, chuẩn hóa lợi thế theo từng cặp, chuẩn
    hóa mục tiêu giá trị và các cơ chế kiểm soát bước cập nhật. Mô hình hội
    tụ ổn định trên quy mô $300$ tác tử sau khoảng $500$ episode.

    \item Đánh giá đối chứng: so sánh IPPO với ba chính sách cổ điển
    (EOQ, $(s, S)$, Newsvendor) đã được tinh chỉnh tham số bằng tìm kiếm lưới,
    theo giao thức đa hạt giống có kiểm định thống kê
    \parencite{henderson2018, agarwal2021}, cả trong so sánh trực tiếp và so
    sánh ở cùng mức phục vụ.
\end{enumerate}

\subsection{Trả lời các câu hỏi nghiên cứu}
\label{subsec:traLoiCauHoi}

Câu hỏi 1 (IPPO so với các baseline cổ điển): so sánh trực tiếp không
kiểm soát mức phục vụ cho thấy IPPO có tổng chi phí cao hơn EOQ khoảng
$3{,}8\%$ nhưng đạt fill rate $85{,}21\%$ so với $75{,}01\%$ của EOQ -- một
so sánh không công bằng vì hai chính sách đang vận hành ở hai mức phục vụ
khác nhau. Khi so sánh ở cùng ngưỡng mức phục vụ tối thiểu ($\ge 84{,}71\%$),
IPPO có tổng chi phí thấp nhất trong cả bốn phương pháp: rẻ hơn EOQ $47{,}5\%$,
rẻ hơn $(s, S)$ $21{,}2\%$ và rẻ hơn Newsvendor $11{,}3\%$
(Mục~\ref{subsec:isoService}). Câu trả lời cho Câu hỏi 1 vì vậy là \textbf{có,
với điều kiện}: IPPO không rẻ hơn tuyệt đối trong mọi phép so sánh, nhưng vượt
trội rõ rệt khi so sánh đúng đắn ở cùng một mức phục vụ -- đúng là điều kiện
so sánh có ý nghĩa kinh tế thực sự.

Câu hỏi 2 (Tổng quát hóa qua parameter sharing): phân tích theo ba
nhóm tam phân vị quy mô cầu (Mục~\ref{subsec:tongQuatHoa}) cho thấy khả năng
tổng quát hóa là \textbf{có điều kiện}. Chính sách chia sẻ tham số đạt fill
rate gần hoặc vượt ngưỡng mục tiêu ở nhóm cầu trung bình ($83{,}57\%$) và cao
($86{,}31\%$), nhưng chỉ đạt $72{,}79\%$ ở nhóm cầu thấp và gián đoạn nhất,
với một số cặp cá biệt xuống dưới $10\%$. Cơ chế chia sẻ tham số vì vậy tổng
quát hóa tốt cho phần lớn hệ thống nhưng chưa đủ để bù đắp hoàn toàn cho các
mặt hàng có đặc trưng nhu cầu cực đoan nhất.

Câu hỏi 3 (Local reward so với global reward): khóa luận triển khai và
đánh giá đầy đủ phương án phần thưởng phân rã cục bộ có chuẩn hóa theo từng
cặp trên toàn bộ thực nghiệm ở Chương~\ref{chap:chapter4}, được củng cố bởi
một quan sát định lượng trong giai đoạn phát triển cho thấy việc thiếu cơ chế
chuẩn hóa từng gây ra hiện tượng không hội tụ (Mục~\ref{subsec:ablationReward}).
Tuy nhiên, một thí nghiệm loại trừ (ablation) có kiểm soát, so sánh trực tiếp
phần thưởng cục bộ với một phần thưởng toàn cục dùng chung trên đúng cấu hình
$300$ cặp cuối cùng, \textbf{chưa được thực hiện} trong phạm vi khóa luận này.
Câu hỏi 3 vì vậy chỉ được trả lời một phần, và được xác định là hạn chế cùng
hướng nghiên cứu tiếp theo quan trọng nhất (Mục~\ref{sec:huongPhatTrien}).

\section{Những hạn chế còn tồn tại}
\label{sec:hanChe}

\begin{itemize}
    \item Cấu trúc mạng lưới đơn giản: các kho hoạt động song song và
    độc lập về nguồn cung; chưa xét cấu trúc đa cấp (multi-echelon) với quan hệ
    cung ứng giữa các kho \parencite{clark1960} và hiệu ứng bullwhip
    \parencite{lee1997}.

    \item Không có truyền thông giữa các tác tử: các tác tử ra quyết
    định hoàn toàn dựa trên quan sát cục bộ, chưa tận dụng thông tin từ các cặp
    kho--mặt hàng khác ngoài tín hiệu mức sử dụng sức chứa kho dùng chung.

    \item Không gian hành động rời rạc: chỉ có $K = 6$ mức đặt
    hàng tương đối theo cầu; với các mặt hàng có nhu cầu rất thấp, bước nhảy
    giữa các mức trở nên thô và góp phần vào hạn chế nêu ở
    Mục~\ref{subsec:tongQuatHoa}.

    \item Chưa tích hợp dự báo nhu cầu: tác tử sử dụng trực tiếp nhu
    cầu lịch sử thay vì dự báo bằng mô hình học sâu chuỗi thời gian.

    \item Số hạt giống thực nghiệm còn khiêm tốn: kết quả đa hạt giống
    được báo cáo trên $3$ hạt giống, thấp hơn khuyến nghị phổ biến ($5$ trở
    lên) do giới hạn thời gian huấn luyện trên CPU.

    \item Thiếu thí nghiệm loại trừ (ablation) trên cấu hình cuối cùng:
    như đã nêu ở Câu hỏi nghiên cứu 3, vai trò của cơ chế chuẩn hóa phần
    thưởng theo từng cặp và phương án phần thưởng toàn cục dùng chung chưa
    được kiểm chứng bằng một thí nghiệm đối chứng trực tiếp trên đúng cấu
    hình $300$ cặp báo cáo ở Chương~\ref{chap:chapter4}.

    \item Thực nghiệm trên môi trường mô phỏng: chưa được kiểm chứng
    trên hệ thống ERP thực tế hoặc dữ liệu vận hành trực tiếp; đơn giá thiếu
    hàng theo giá thực tế còn dựa trên một giả định về tỷ lệ biên lợi nhuận
    ($30\%$) do bộ dữ liệu M5 không công bố giá vốn (Mục~\ref{subsec:thamSoChiPhi}).
\end{itemize}

\section{Hướng phát triển trong tương lai}
\label{sec:huongPhatTrien}

\begin{itemize}
    \item Hoàn thiện thí nghiệm loại trừ cho Câu hỏi nghiên cứu 3:
    huấn luyện lại trên đúng cấu hình $300$ cặp hiện tại với cơ chế chuẩn hóa
    phần thưởng theo từng cặp bị tắt, và với một phương án phần thưởng toàn
    cục dùng chung, cùng hạt giống và số episode, để có bằng chứng đối chứng
    trực tiếp thay vì chỉ dựa trên quan sát phát triển sơ bộ.

    \item Mở rộng số hạt giống đánh giá: chạy thêm hạt giống để đạt tối
    thiểu $5$ hạt giống độc lập theo khuyến nghị về tính tái lập trong học
    tăng cường sâu.

    \item Cải thiện chính sách cho nhóm SKU nhu cầu thấp: xem xét
    không gian hành động liên tục hoặc bảng mức đặt hàng có độ phân giải mịn
    hơn cho các cặp có nhu cầu rất thấp, nhằm thu hẹp khoảng cách fill rate đã
    ghi nhận ở nhóm này (Mục~\ref{subsec:tongQuatHoa}).

    \item Mở rộng sang cấu trúc đa cấp (multi-echelon): mô hình hóa
    quan hệ cung ứng giữa kho trung tâm và kho khu vực, cho phép tác tử học
    chiến lược điều phối liên kho.

    \item Tích hợp cơ chế truyền thông giữa tác tử: áp dụng các
    phương pháp như CommNet hoặc TarMAC để cho phép tác tử chia sẻ thông tin,
    đặc biệt về tình trạng sức chứa kho.

    \item Tích hợp dự báo nhu cầu: nhúng mô hình dự báo chuỗi thời
    gian (LSTM, Transformer) vào vector quan sát của tác tử, tận dụng thông tin
    xu hướng và mùa vụ.

    \item Kiểm chứng giả định kinh tế về đơn giá thiếu hàng: thu thập
    hoặc ước lượng dữ liệu giá vốn thực tế để thay thế giả định tỷ lệ biên lợi
    nhuận hiện tại, và phân tích độ nhạy của kết quả theo tỷ lệ này.

    \item Mở rộng quy mô: thử nghiệm với số lượng SKU lớn hơn trong bộ
    dữ liệu M5 gốc và đánh giá lại khả năng mở rộng của cơ chế chia sẻ tham số
    khi quy mô hệ thống tăng lên.

    \item Triển khai thử nghiệm thực tế: xây dựng hệ thống dạng SaaS
    kết nối API với nền tảng quản lý kho, cho phép đánh giá mô hình trên dữ liệu
    vận hành trực tiếp.
\end{itemize}
